cmake_minimum_required(VERSION 3.28)
project(CppChatServers LANGUAGES CXX)

set(CPPCHAT_SOURCE_DIR "${PROJECT_SOURCE_DIR}")
set(CPPCHAT_BINARY_DIR "${PROJECT_BINARY_DIR}")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
option(CPPCHAT_BUILD_TESTS "Build isolated regression tests" OFF)
set(CPPCHAT_SANITIZER "" CACHE STRING "Linux sanitizer: address, undefined, or thread")
option(CPPCHAT_COVERAGE "Instrument core logic for coverage" OFF)
include(cmake/TestInstrumentation.cmake)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT GateServer)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
set_property(GLOBAL PROPERTY PREDEFINED_TARGETS_FOLDER "构建辅助")

include(cmake/ServerDependencies.cmake)
include(cmake/ServerLogging.cmake)
include(cmake/ServerTargets.cmake)

add_subdirectory(GateServer)
add_subdirectory(StatusServer)
add_subdirectory(ChatServer)
add_subdirectory(ChatServer2)
add_subdirectory(ResourceServer)
if(CPPCHAT_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests/cpp)
endif()

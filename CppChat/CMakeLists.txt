cmake_minimum_required(VERSION 3.24)
project(CppChat VERSION 0.1.6 LANGUAGES C CXX)
set(CPPCHAT_APP_VERSION "${PROJECT_VERSION}" CACHE STRING "Client and installer release version")
if(NOT CPPCHAT_APP_VERSION MATCHES "^(0|[1-9][0-9]?[0-9]?[0-9]?[0-9]?)\\.(0|[1-9][0-9]?[0-9]?[0-9]?[0-9]?)\\.(0|[1-9][0-9]?[0-9]?[0-9]?[0-9]?)$")
  message(FATAL_ERROR "CPPCHAT_APP_VERSION must be a three-part numeric version")
endif()
option(CPPCHAT_BUILD_TESTS "Build native Qt regression tests" OFF)
get_filename_component(repository_root "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
if(CMAKE_CURRENT_BINARY_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR OR CMAKE_CURRENT_BINARY_DIR STREQUAL repository_root)
  message(FATAL_ERROR "Build the client in CppChat/build/debug or CppChat/build/release, outside the source directories.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC_SEARCH_PATHS "${CMAKE_CURRENT_SOURCE_DIR}/app" "${CMAKE_CURRENT_SOURCE_DIR}/dialogs")
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Sql Svg Concurrent)
include(cmake/EncryptedSqlite.cmake)
file(GLOB_RECURSE client_sources CONFIGURE_DEPENDS app/*.cpp app/*.h app/*.ui common/*.cpp common/*.h dialogs/*.cpp dialogs/*.h dialogs/*.ui network/*.cpp network/*.h services/*.cpp services/*.h widgets/*.cpp widgets/*.h)
list(REMOVE_ITEM client_sources "${CMAKE_CURRENT_SOURCE_DIR}/app/main.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/common/securitypolicy.cpp")
add_library(CppChatCore OBJECT ${client_sources} rc.qrc)
qt_add_executable(CppChat WIN32 app/main.cpp common/securitypolicy.cpp)
target_compile_definitions(CppChat PRIVATE $<$<CONFIG:Debug>:CPPCHAT_DEVELOPMENT_TRANSPORT>)
add_dependencies(CppChat CppChatSqliteCipher)
target_link_libraries(CppChat PRIVATE CppChatCore)
source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${client_sources} "${CMAKE_CURRENT_SOURCE_DIR}/rc.qrc")
target_include_directories(CppChatCore PUBLIC . app common dialogs network services widgets third_party/spdlog/include)
target_compile_definitions(CppChatCore PUBLIC CPPCHAT_VERSION="${CPPCHAT_APP_VERSION}")
target_link_libraries(CppChatCore PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Sql Qt6::Svg Qt6::Concurrent)
if(WIN32)
  target_sources(CppChat PRIVATE app/app.rc)
  set_property(SOURCE app/app.rc APPEND PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Whatsapp.ico")
  target_link_libraries(CppChatCore PUBLIC dwmapi crypt32 advapi32 bcrypt)
endif()
if(MINGW)
  target_compile_options(CppChatCore PRIVATE $<$<CONFIG:Debug>:-g3> $<$<CONFIG:Debug>:-Og>)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/config.ini")
  configure_file(config.ini "${CMAKE_CURRENT_BINARY_DIR}/config.ini" COPYONLY)
endif()

option(CPPCHAT_DEPLOY_QT "Deploy Qt libraries and plugins beside the client executable" ON)
if(WIN32 AND CPPCHAT_DEPLOY_QT)
  get_filename_component(qt_bin "${Qt6_DIR}/../../../bin" ABSOLUTE)
  find_program(WINDEPLOYQT_EXECUTABLE NAMES windeployqt HINTS "${qt_bin}" NO_DEFAULT_PATH REQUIRED)
  if(MINGW)
    # This MinGW Qt distribution provides one DLL set for Debug and Release apps.
    set(deploy_configuration --release)
  else()
    set(deploy_configuration "$<IF:$<CONFIG:Debug>,--debug,--release>")
  endif()
  add_custom_command(TARGET CppChat POST_BUILD
    COMMAND "${WINDEPLOYQT_EXECUTABLE}" ${deploy_configuration}
      --no-translations --no-system-d3d-compiler --no-opengl-sw "$<TARGET_FILE:CppChat>"
    COMMENT "Deploying the selected Qt Kit's runtime for CppChat" VERBATIM)
endif()
if(CPPCHAT_BUILD_TESTS)
  enable_testing()
  add_subdirectory("${repository_root}/tests/qt" "tests/qt")
endif()
